// Helper para formatar números no padrão BR com pontos (ex: 1.448.212,40)
function formatarMoedaBR(valor) {
  const num = parseFloat(valor) || 0;
  return num.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
}

// Helper para converter texto digitado de volta para float
function converterTextoParaNumero(str) {
  if (!str) return 0;
  const limpo = String(str).replace(/R\$\s?/g, '').replace(/\./g, '').replace(',', '.').trim();
  return parseFloat(limpo) || 0;
}

// SOMA EM TEMPO REAL PARA OS CARDS DO TOPO
// ==========================================
function atualizarTotaisTopo() {
  let totalOrcado = 0;
  let totalRealizado = 0;

  document.querySelectorAll('.input-budget-val').forEach(input => {
    totalOrcado += converterTextoParaNumero(input.value);
  });

  document.querySelectorAll('.input-realizado-val').forEach(input => {
    totalRealizado += converterTextoParaNumero(input.value);
  });

  const elOrcado = document.getElementById('orcamento-total-mes') || document.getElementById('orcamento-total');
  const elRealizado = document.getElementById('receita-realizada-mes') || document.getElementById('receita-realizada');

  if (elOrcado) elOrcado.value = "R$ " + formatarMoedaBR(totalOrcado);
  if (elRealizado) elRealizado.value = "R$ " + formatarMoedaBR(totalRealizado);
}

function popularSelectsDepartamentos() {

function inicializarSeletoresMes() {
  const hoje = new Date();
  const mesAtual = String(hoje.getMonth() + 1).padStart(2, '0');
  const anoAtual = hoje.getFullYear();
  const valorMesAno = `${anoAtual}-${mesAtual}`;

  const selectBudget = document.getElementById('budget-mes-ano');
  if (selectBudget && !selectBudget.value) {
    selectBudget.value = valorMesAno;
  }
}

function obterChaveMes(dataString) {
  const partes = dataString.split('-');
  return `${partes[1]}-${partes[0]}`;
}

function mudarAba(aba) {
  const btnLanc = document.getElementById('btn-tab-lancamentos');
  const btnBudg = document.getElementById('btn-tab-budgets');
  const btnDash = document.getElementById('btn-tab-dashboard');

  const abaLanc = document.getElementById('aba-lancamentos');
  const abaBudg = document.getElementById('aba-budgets');
  const abaDash = document.getElementById('aba-dashboard');

  [btnLanc, btnBudg, btnDash].forEach(b => {
    if (b) b.className = "px-3 py-1.5 rounded-lg text-sm font-medium hover:bg-hotel-dark text-emerald-100 transition";
  });

  if (abaLanc) abaLanc.classList.add('hidden');
  if (abaBudg) abaBudg.classList.add('hidden');
  if (abaDash) abaDash.classList.add('hidden');

  if (aba === 'lancamentos') {
    if (abaLanc) abaLanc.classList.remove('hidden');
    if (btnLanc) btnLanc.className = "px-3 py-1.5 rounded-lg text-sm font-medium bg-white text-hotel-primary shadow transition";
  } else if (aba === 'budgets') {
    if (abaBudg) abaBudg.classList.remove('hidden');
    if (btnBudg) btnBudg.className = "px-3 py-1.5 rounded-lg text-sm font-medium bg-white text-hotel-primary shadow transition";

  } else {
    if (abaDash) abaDash.classList.remove('hidden');
    if (btnDash) btnDash.className = "px-3 py-1.5 rounded-lg text-sm font-medium bg-white text-hotel-primary shadow transition";
    atualizarGraficos();
  }
}

function renderizarFormBudgets() {
  inicializarSeletoresMes();
  const mesAnoInput = document.getElementById('budget-mes-ano').value;
  if (!mesAnoInput) return;

  const chaveMes = obterChaveMes(mesAnoInput);
  const budgetsDoMes = budgets[chaveMes] || {};
  const realizadasDoMes = receitasRealizadas[chaveMes] || {};

  const partesMes = mesAnoInput.split('-');
  const anoSel = parseInt(partesMes[0]);
  const mesSel = parseInt(partesMes[1]) - 1;

  const custosAcumuladosPorDepto = {};
  lancamentos.forEach(item => {
    const p = item.data.split('-');
    const a = parseInt(p[0]);
    const m = parseInt(p[1]) - 1;

    if (a === anoSel && m === mesSel) {
      custosAcumuladosPorDepto[item.departamento] = (custosAcumuladosPorDepto[item.departamento] || 0) + item.valor;
    }
  });

  const container = document.getElementById('lista-inputs-budget');
  if (!container) return;
  container.innerHTML = '';

  departamentosLista.forEach(depto => {
    const valOrcado = budgetsDoMes[depto] || 0;
    const valRealizado = (typeof realizadasDoMes === 'object' && realizadasDoMes[depto]) ? realizadasDoMes[depto] : 0;
    const valCustoRequisicoes = custosAcumuladosPorDepto[depto] || 0;

    const percCusto = valRealizado > 0 ? ((valCustoRequisicoes / valRealizado) * 100).toFixed(1) : '0.0';
    const corBadgePerc = parseFloat(percCusto) > 29.0 ? 'bg-red-100 text-red-700 font-bold' : 'bg-emerald-100 text-emerald-800 font-bold';

    const div = document.createElement('div');
    div.className = "bg-gray-50 p-2.5 rounded-lg border border-gray-200 grid grid-cols-1 md:grid-cols-12 gap-2 items-center";
    div.innerHTML = `
      <div class="md:col-span-4 font-semibold text-xs text-gray-800 flex justify-between md:justify-start items-center gap-1">
        <span>${depto}</span>
        <span class="md:hidden text-[10px] px-1.5 py-0.5 rounded ${corBadgePerc}">${percCusto}% Custo</span>
      </div>

      <!-- Coluna 1: Receita Orçada -->
      <div class="md:col-span-3 relative">
        <span class="absolute left-2.5 top-2 text-xs font-semibold text-gray-400">R$</span>
        <input type="text" data-depto="${depto}" value="${formatarMoedaBR(valOrcado)}" 
               oninput="atualizarTotaisTopo()"
               class="input-budget-val w-full pl-8 pr-2 py-1.5 border border-gray-300 rounded text-xs font-bold text-gray-800 focus:ring-2 focus:ring-[#00463f] focus:outline-none">
      </div>

      <!-- Coluna 2: Receita Realizada -->
      <div class="md:col-span-3 relative">
        <span class="absolute left-2.5 top-2 text-xs font-semibold text-blue-400">R$</span>
        <input type="text" data-depto-real="${depto}" value="${formatarMoedaBR(valRealizado)}" 
               oninput="atualizarTotaisTopo()"
               class="input-realizado-val w-full pl-8 pr-2 py-1.5 border border-blue-200 bg-blue-50/40 rounded text-xs font-bold text-blue-900 focus:ring-2 focus:ring-blue-600 focus:outline-none">
      </div>

      <!-- Coluna 3: Custos Acumulados -->
      <div class="md:col-span-2 bg-amber-50/80 p-1.5 rounded border border-amber-200 text-center flex flex-col justify-center">
        <span class="text-xs font-extrabold text-amber-900">R$ ${formatarMoedaBR(valCustoRequisicoes)}</span>
        <span class="text-[10px] ${corBadgePerc} px-1 rounded-full mt-0.5 self-center">${percCusto}% da Rec. Real</span>
      </div>
    `;
    container.appendChild(div);
  });

  // Atualiza automaticamente os cards do topo
  atualizarTotaisTopo();
}

function salvarBudgets(e) {
  e.preventDefault();
  const mesAnoInput = document.getElementById('budget-mes-ano').value;
  const chaveMes = obterChaveMes(mesAnoInput);

  if (!budgets[chaveMes]) budgets[chaveMes] = {};
  if (!receitasRealizadas[chaveMes]) receitasRealizadas[chaveMes] = {};

  const inputsOrcado = document.querySelectorAll('.input-budget-val');
  inputsOrcado.forEach(input => {
    const depto = input.getAttribute('data-depto');
    budgets[chaveMes][depto] = Math.round(converterTextoParaNumero(input.value) * 100) / 100;
  });

  const inputsRealizado = document.querySelectorAll('.input-realizado-val');
  inputsRealizado.forEach(input => {
    const depto = input.getAttribute('data-depto-real');
    if (typeof receitasRealizadas[chaveMes] !== 'object') {
      receitasRealizadas[chaveMes] = {};
    }
    receitasRealizadas[chaveMes][depto] = Math.round(converterTextoParaNumero(input.value) * 100) / 100;
  });

  localStorage.setItem('requisicoes_budgets', JSON.stringify(budgets));
  localStorage.setItem('requisicoes_receitas_realizadas', JSON.stringify(receitasRealizadas));

  renderizarFormBudgets();
  alert(`✅ Configurações salvas com sucesso para ${chaveMes}!`);
}

// MÓDULO DE IMPORTAÇÃO DA DRE (.XLSX)
// ==========================================
function importarRelatorioDRE(evento) {
  const file = evento.target.files[0];
  if (!file) return;

  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const data = new Uint8Array(e.target.result);
      const workbook = XLSX.read(data, { type: 'array' });
      const sheet = workbook.Sheets[workbook.SheetNames[0]];
      
      const rows = XLSX.utils.sheet_to_json(sheet, { header: 1, raw: true });

      const mesAnoInput = document.getElementById('budget-mes-ano').value;
      const chaveMes = obterChaveMes(mesAnoInput);

      budgets[chaveMes] = {};
      receitasRealizadas[chaveMes] = {};

      const converterParaNumero = (val) => {
        if (typeof val === 'number') return isNaN(val) ? 0 : val;
        if (!val) return 0;
        const textoLimpo = String(val).replace(/R\$\s?/g, '').replace(/\./g, '').replace(',', '.').trim();
        const num = parseFloat(textoLimpo);
        return isNaN(num) ? 0 : num;
      };

      let blocoAlimentosAtivo = false;

      rows.forEach((row) => {
        if (!row || row.length < 1) return;

        const deptoTexto = row[0] ? String(row[0]).trim() : '';
        const deptoLower = deptoTexto.toLowerCase();

        // 1. LIGA a leitura apenas em contas exclusivas de ALIMENTOS
        if (
          deptoTexto.includes('3.11.102.001') || deptoLower.includes('receitas de alimentos') ||
          deptoTexto.includes('3.11.102.003') || deptoLower.includes('cafe da manha') || deptoLower.includes('café da manhã') ||
          deptoTexto.includes('3.11.102.004') || deptoLower.includes('map / fap - alimentos')
        ) {
          blocoAlimentosAtivo = true;
          return;
        }

        // 2. DESLIGA a leitura em qualquer conta de BEBIDAS ou Impostos/Deduções
        if (
          deptoTexto.includes('3.11.102.002') || deptoLower.includes('receitas de bebidas') ||
          deptoTexto.includes('3.11.102.005') || deptoLower.includes('map / fap - bebidas') ||
          deptoTexto.includes('3.11.107') || deptoLower.includes('deducoes') || deptoLower.includes('deduções') ||
          deptoLower.includes('impostos')
        ) {
          blocoAlimentosAtivo = false;
        }

        // 3. Processa e soma os valores apenas quando o bloco ativo for ALIMENTOS
        if (blocoAlimentosAtivo) {
          const match = deptoTexto.match(/24\d{2}/);
          if (match) {
            const codigoDRE = match[0];
            const nomeLimpo = mapeamentoDeptos[codigoDRE];

            if (nomeLimpo) {
              const orcadoVal = converterParaNumero(row[2]);
              const realizadoVal = converterParaNumero(row[4]);

              const orcadoAtual = budgets[chaveMes][nomeLimpo] || 0;
              const realizadoAtual = receitasRealizadas[chaveMes][nomeLimpo] || 0;

              budgets[chaveMes][nomeLimpo] = Math.round((orcadoAtual + orcadoVal) * 100) / 100;
              receitasRealizadas[chaveMes][nomeLimpo] = Math.round((realizadoAtual + realizadoVal) * 100) / 100;
            }
          }
        }
      });

      localStorage.setItem('requisicoes_budgets', JSON.stringify(budgets));
      localStorage.setItem('requisicoes_receitas_realizadas', JSON.stringify(receitasRealizadas));

      renderizarFormBudgets();
      
      let totalRealizadoFinal = 0;
      departamentosLista.forEach(d => totalRealizadoFinal += (receitasRealizadas[chaveMes][d] || 0));
      
      alert(`✅ DRE (Apenas Alimentos) importada com sucesso para ${chaveMes}!\nReceita Realizada Total: R$ ${formatarMoedaBR(totalRealizadoFinal)}`);

    } catch (err) {
      console.error(err);
      alert('⚠️ Ocorreu um erro ao processar o arquivo da DRE.');
    }
  };
  reader.readAsArrayBuffer(file);
}

document.getElementById('form-requisicao').addEventListener('submit', (e) => {
  e.preventDefault();

  const index = parseInt(document.getElementById('edit-index').value);
  const dataVal = document.getElementById('data').value;
  const deptoVal = document.getElementById('departamento').value;
  const valorVal = parseFloat(document.getElementById('valor').value);

  const chaveMes = obterChaveMes(dataVal);
  
  let receitaBase = 0;
  if (receitasRealizadas[chaveMes] && typeof receitasRealizadas[chaveMes] === 'object' && receitasRealizadas[chaveMes][deptoVal]) {
    receitaBase = receitasRealizadas[chaveMes][deptoVal];
  } else if (budgets[chaveMes] && budgets[chaveMes][deptoVal]) {
    receitaBase = budgets[chaveMes][deptoVal];
  }

  const teto = receitaBase * 0.29;

  if (teto > 0) {
    const partesNovaData = dataVal.split('-');
    const mesNovo = parseInt(partesNovaData[1]) - 1;
    const anoNovo = parseInt(partesNovaData[0]);

    const totalMesAtual = lancamentos.reduce((acc, item, idx) => {
      if (idx === index) return acc;
      const p = item.data.split('-');
      const m = parseInt(p[1]) - 1;
      const a = parseInt(p[0]);
      if (item.departamento === deptoVal && m === mesNovo && a === anoNovo) {
        return acc + item.valor;
      }
      return acc;
    }, 0);

    if (totalMesAtual + valorVal > teto) {
      const excesso = (totalMesAtual + valorVal) - teto;
      alert(`⚠️ ATENÇÃO: Este lançamento excede o limite permitido (29% da receita) do setor (${deptoVal}) em R$ ${formatarMoedaBR(excesso)}!\nO lançamento será gravado para fins de registro.`);
    }
  }

  const partes = dataVal.split('-');
  const dataObj = new Date(partes[0], partes[1] - 1, partes[2]);

  const novoItem = {
    data: dataVal,
    diaSemanaIndex: dataObj.getDay(),
    departamento: deptoVal,
    valor: valorVal
  };

  if (index === -1) {
    lancamentos.push(novoItem);
  } else {
    lancamentos[index] = novoItem;
  }

  localStorage.setItem('requisicoes_dados', JSON.stringify(lancamentos));
  limparFormulario();
  filtrarLancamentos();
});

function obterIntervaloSemana(dataRef) {
  const d = new Date(dataRef);
  const diaSemana = d.getDay();
  const diffSegunda = d.getDate() - (diaSemana === 0 ? 6 : diaSemana - 1);
  
  const inicioSegunda = new Date(d.setDate(diffSegunda));
  inicioSegunda.setHours(0,0,0,0);
  
  const fimDomingo = new Date(inicioSegunda);
  fimDomingo.setDate(inicioSegunda.getDate() + 6);
  fimDomingo.setHours(23,59,59,999);

  return { inicio: inicioSegunda, fim: fimDomingo };
}

function alternarModoFiltro() {
  const modo = document.getElementById('filtro-modo').value;
  const inputDataBase = document.getElementById('filtro-data-base');

  if (modo === 'semana_especifica') {
    inputDataBase.classList.remove('hidden');
    if (!inputDataBase.value) inputDataBase.valueAsDate = new Date();
  } else {
    inputDataBase.classList.add('hidden');
  }
  filtrarLancamentos();
}

function filtrarLancamentos() {
  const modo = document.getElementById('filtro-modo').value;
  const deptoFiltro = document.getElementById('filtro-depto').value;
  const inputDataBase = document.getElementById('filtro-data-base').value;
  const infoSemana = document.getElementById('info-semana-atual');

  let dataRef = new Date();
  if (modo === 'semana_especifica' && inputDataBase) {
    const p = inputDataBase.split('-');
    dataRef = new Date(p[0], p[1] - 1, p[2]);
  }

  const { inicio, fim } = obterIntervaloSemana(dataRef);

  const lancamentosFiltrados = lancamentos.filter((item, index) => {
    item._originalIndex = index;
    const partes = item.data.split('-');
    const itemData = new Date(partes[0], partes[1] - 1, partes[2]);

    let passaData = true;
    if (modo === 'semana_atual' || modo === 'semana_especifica') {
      passaData = itemData >= inicio && itemData <= fim;
    } else if (modo === 'mes_atual') {
      const hoje = new Date();
      passaData = itemData.getMonth() === hoje.getMonth() && itemData.getFullYear() === hoje.getFullYear();
    }

    let passaDepto = (deptoFiltro === 'todos') || (item.departamento === deptoFiltro);

    return passaData && passaDepto;
  });

  if (modo === 'semana_atual' || modo === 'semana_especifica') {
    infoSemana.innerText = `Exibindo semana de ${inicio.toLocaleDateString('pt-BR')} (Seg) a ${fim.toLocaleDateString('pt-BR')} (Dom)`;
  } else if (modo === 'mes_atual') {
    infoSemana.innerText = `Exibindo lançamentos do mês atual`;
  } else {
    infoSemana.innerText = `Exibindo todo o histórico registrado`;
  }

  renderizarTabela(lancamentosFiltrados);
}

function renderizarTabela(dadosExibicao) {
  const tbody = document.getElementById('tabela-corpo');
  const totalFiltroEl = document.getElementById('total-filtro');
  tbody.innerHTML = '';

  let totalAcumulado = 0;

  if (dadosExibicao.length === 0) {
    tbody.innerHTML = `<tr><td colspan="5" class="p-4 text-center text-gray-500">Nenhum lançamento encontrado para este período.</td></tr>`;
    totalFiltroEl.innerText = "R$ 0,00";
    return;
  }

  dadosExibicao.sort((a, b) => new Date(b.data) - new Date(a.data));

  dadosExibicao.forEach((item) => {
    totalAcumulado += item.valor;
    const tr = document.createElement('tr');
    tr.className = "hover:bg-emerald-50/50 transition";

    const dataFormatada = item.data.split('-').reverse().join('/');
    const valorFormatado = "R$ " + formatarMoedaBR(item.valor);

    tr.innerHTML = `
      <td class="p-2 font-medium text-gray-700">${dataFormatada}</td>
      <td class="p-2 text-gray-600">${diasNomes[item.diaSemanaIndex]}</td>
      <td class="p-2 font-semibold text-hotel-primary">${item.departamento}</td>
      <td class="p-2 font-bold text-gray-800">${valorFormatado}</td>
      <td class="p-2 text-center space-x-1">
        <button onclick="prepararEdicao(${item._originalIndex})" class="px-2 py-0.5 bg-amber-500 text-white text-[11px] rounded hover:bg-amber-600 font-medium">Editar</button>
        <button onclick="excluirItem(${item._originalIndex})" class="px-2 py-0.5 bg-red-600 text-white text-[11px] rounded hover:bg-red-700 font-medium">Excluir</button>
      </td>
    `;
    tbody.appendChild(tr);
  });

  totalFiltroEl.innerText = "R$ " + formatarMoedaBR(totalAcumulado);
}

function prepararEdicao(indexOriginal) {
  const item = lancamentos[indexOriginal];
  document.getElementById('edit-index').value = indexOriginal;
  document.getElementById('data').value = item.data;
  document.getElementById('departamento').value = item.departamento;
  document.getElementById('valor').value = item.valor;

  document.getElementById('form-titulo').innerHTML = "<span>✏️</span> Editar Lançamento";
  document.getElementById('btn-salvar').innerText = "Atualizar";
  document.getElementById('btn-cancelar').classList.remove('hidden');
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

function excluirItem(indexOriginal) {
  if (confirm("Deseja realmente excluir este lançamento?")) {
    lancamentos.splice(indexOriginal, 1);
    localStorage.setItem('requisicoes_dados', JSON.stringify(lancamentos));
    filtrarLancamentos();
  }
}

function limparFormulario() {
  document.getElementById('form-requisicao').reset();
  document.getElementById('edit-index').value = "-1";
  document.getElementById('data').valueAsDate = new Date();
  document.getElementById('form-titulo').innerHTML = "<span>📝</span> Novo Lançamento";
  document.getElementById('btn-salvar').innerText = "Salvar";
  document.getElementById('btn-cancelar').classList.add('hidden');
}

function formatarK(valor) {
  if (valor === 0) return 'R$ 0';
  if (valor >= 1000) {
    return 'R$ ' + (valor / 1000).toLocaleString('pt-BR', { maximumFractionDigits: 1 }) + 'k';
  }
  return 'R$ ' + valor.toLocaleString('pt-BR');
}

function atualizarGraficos() {
  const modoDash = document.getElementById('dash-filtro-periodo').value;
  const { inicio, fim } = obterIntervaloSemana(new Date());

  const dadosDash = lancamentos.filter(item => {
    const partes = item.data.split('-');
    const itemData = new Date(partes[0], partes[1] - 1, partes[2]);

    if (modoDash === 'semana_atual') {
      return itemData >= inicio && itemData <= fim;
    } else if (modoDash === 'mes_atual') {
      const hoje = new Date();
      return itemData.getMonth() === hoje.getMonth() && itemData.getFullYear() === hoje.getFullYear();
    }
    return true;
  });

  const totaisPorDepto = {};
  let totalGeral = 0;

  dadosDash.forEach(item => {
    totalGeral += item.valor;
    totaisPorDepto[item.departamento] = (totaisPorDepto[item.departamento] || 0) + item.valor;
  });

  const hoje = new Date();
  const chaveMesAtual = `${String(hoje.getMonth() + 1).padStart(2, '0')}-${hoje.getFullYear()}`;
  const budgetsMesAtual = budgets[chaveMesAtual] || {};
  const realizadasMesAtual = receitasRealizadas[chaveMesAtual] || {};

  let receitaRealizadaHotel = 0;
  let receitaOrcadaHotel = 0;

  departamentosLista.forEach(d => {
    receitaOrcadaHotel += (budgetsMesAtual[d] || 0);
    receitaRealizadaHotel += (realizadasMesAtual[d] || 0);
  });

  document.getElementById('dash-card-total').innerText = "R$ " + formatarMoedaBR(totalGeral);
  
  const elPerc = document.getElementById('dash-card-percentual');
  const receitaBase = receitaRealizadaHotel > 0 ? receitaRealizadaHotel : receitaOrcadaHotel;

  if (receitaBase > 0) {
    const percMetaProporcional = ((totalGeral / receitaBase) * 100).toFixed(1);
    elPerc.innerText = `${percMetaProporcional}%`;
    elPerc.className = `text-xl lg:text-2xl font-extrabold mt-0.5 ${percMetaProporcional > 29.0 ? 'text-red-600' : 'text-emerald-700'}`;
  } else {
    elPerc.innerText = 'N/A';
    elPerc.className = 'text-xl lg:text-2xl font-extrabold text-gray-400 mt-0.5';
  }

  let topDepto = '-';
  let maiorValor = 0;
  for (let depto in totaisPorDepto) {
    if (totaisPorDepto[depto] > maiorValor) {
      maiorValor = totaisPorDepto[depto];
      topDepto = depto;
    }
  }
  document.getElementById('dash-card-top-depto').innerText = topDepto;

  const labelsDeptos = [];
  const valoresGastos = [];
  const valoresTetos = [];
  const coresGastos = [];
  const percRealizadoDeptos = [];

  const setoresEstourados = [];
  const setoresAtencao = [];
  const setoresComFolga = [];

  departamentosLista.forEach(depto => {
    const gasto = totaisPorDepto[depto] || 0;
    const recReal = (typeof realizadasMesAtual === 'object' && realizadasMesAtual[depto]) ? realizadasMesAtual[depto] : (budgetsMesAtual[depto] || 0);
    const teto = recReal * 0.29;

    if (gasto > 0 || teto > 0) {
      labelsDeptos.push(depto);
      valoresGastos.push(gasto);
      valoresTetos.push(teto);

      const percRealizado = recReal > 0 ? ((gasto / recReal) * 100) : 0;
      percRealizadoDeptos.push(percRealizado.toFixed(1));

      if (teto > 0 && gasto > teto) {
        coresGastos.push('#dc2626');
        setoresEstourados.push({ depto, gasto, teto, excesso: gasto - teto, perc: percRealizado });
      } else if (teto > 0 && percRealizado >= 23.2) {
        coresGastos.push('#f59e0b');
        setoresAtencao.push({ depto, gasto, teto, perc: percRealizado });
      } else {
        coresGastos.push('#10b981');
        if (teto > 0) {
          setoresComFolga.push({ depto, gasto, teto, saldo: teto - gasto, perc: percRealizado });
        }
      }
    }
  });

  if (chartDeptos) chartDeptos.destroy();
  chartDeptos = new Chart(document.getElementById('chartDeptos'), {
    type: 'bar',
    data: {
      labels: labelsDeptos,
      datasets: [
        {
          label: 'Gasto Real (R$)',
          data: valoresGastos,
          backgroundColor: coresGastos,
          borderRadius: 4,
          barPercentage: 0.85,
          categoryPercentage: 0.9
        },
        {
          label: 'Teto 29% (R$)',
          data: valoresTetos,
          backgroundColor: '#9ca3af',
          borderRadius: 4,
          barPercentage: 0.85,
          categoryPercentage: 0.9
        }
      ]
    },
    options: {
      indexAxis: 'y',
      responsive: true,
      maintainAspectRatio: false,
      plugins: {
        legend: { display: true, position: 'top' },
        datalabels: {
          color: (ctx) => ctx.datasetIndex === 0 ? '#ffffff' : '#1f2937',
          font: { weight: 'bold', size: 11 },
          anchor: (ctx) => ctx.datasetIndex === 0 ? 'start' : 'end',
          align: (ctx) => ctx.datasetIndex === 0 ? 'end' : 'end',
          offset: 6,
          formatter: (value, ctx) => {
            if (value === 0) return '';
            if (ctx.datasetIndex === 0) {
              const p = percRealizadoDeptos[ctx.dataIndex];
              return `${p}% (${formatarK(value)})`;
            }
            return formatarK(value);
          }
        }
      },
      scales: {
        x: {
          ticks: { callback: (val) => 'R$ ' + val.toLocaleString('pt-BR'), font: { size: 10 } },
          grid: { color: '#f3f4f6' }
        },
        y: {
          ticks: { font: { size: 10, weight: 'bold' } },
          grid: { display: false }
        }
      }
    }
  });

  renderizarInsightsIA(setoresEstourados, setoresAtencao, setoresComFolga, receitaOrcadaHotel, receitaRealizadaHotel);
}

function renderizarInsightsIA(estourados, atencao, folga, receitaOrcada, receitaRealizada) {
  const container = document.getElementById('container-ia-insights');
  if (!container) return;

  container.innerHTML = '';
  folga.sort((a, b) => b.saldo - a.saldo);

  if (receitaRealizada > 0 && receitaOrcada > 0) {
    const percAtingimento = ((receitaRealizada / receitaOrcada) * 100).toFixed(1);
    const cardReceita = document.createElement('div');
    cardReceita.className = "p-2.5 rounded-lg bg-blue-50 border border-blue-200 text-xs text-blue-900 shadow-sm mb-2";
    cardReceita.innerHTML = `
      <div class="flex justify-between items-center font-bold">
        <span>📊 RECEITA REALIZADA (DRE)</span>
        <span class="text-[10px] bg-blue-200 text-blue-800 px-1.5 py-0.5 rounded">${percAtingimento}% da Meta</span>
      </div>
      <p class="mt-1 text-[11px]">Realizado: <strong>R$ ${formatarMoedaBR(receitaRealizada)}</strong> | Orçado: R$ ${formatarMoedaBR(receitaOrcada)}</p>
    `;
    container.appendChild(cardReceita);
  }

  if (estourados.length > 0) {
    estourados.forEach(item => {
      let sugestaoHTML = '';
      if (folga.length > 0) {
        const doador = folga[0];
        sugestaoHTML = `
          <div class="mt-2 pt-2 border-t border-red-200/60 text-[11px] text-red-900 font-medium">
            💡 <strong>Ação Recomendada:</strong> Bloquear requisições em <strong>${item.depto}</strong> e remanejar até 
            <strong>R$ ${formatarMoedaBR(item.excesso)}</strong> do saldo disponível de <strong>${doador.depto}</strong> (Sobra: R$ ${formatarMoedaBR(doador.saldo)}).
          </div>
        `;
      } else {
        sugestaoHTML = `
          <div class="mt-2 pt-2 border-t border-red-200/60 text-[11px] text-red-900 font-medium">
            ⛔ <strong>Ação Recomendada:</strong> Cessar novos pedidos imediatamente. Não há saldo excedente disponível em outros setores neste mês.
          </div>
        `;
      }

      const card = document.createElement('div');
      card.className = "p-3 rounded-lg bg-red-50 border border-red-200 text-xs text-red-800 shadow-sm";
      card.innerHTML = `
        <div class="flex justify-between items-start">
          <span class="font-bold text-red-700 flex items-center gap-1">🚨 CESSAR REQUISIÇÕES</span>
          <span class="text-[10px] bg-red-200 text-red-800 px-1.5 py-0.5 rounded font-bold">${item.perc.toFixed(1)}% do Custo</span>
        </div>
        <p class="mt-1 font-semibold text-gray-800">
          O setor <strong>${item.depto}</strong> excedeu o limite de 29% em <strong>R$ ${formatarMoedaBR(item.excesso)}</strong>.
        </p>
        ${sugestaoHTML}
      `;
      container.appendChild(card);
    });
  }

  if (atencao.length > 0) {
    atencao.forEach(item => {
      const card = document.createElement('div');
      card.className = "p-3 rounded-lg bg-amber-50 border border-amber-200 text-xs text-amber-900 shadow-sm";
      card.innerHTML = `
        <div class="flex justify-between items-start">
          <span class="font-bold text-amber-800 flex items-center gap-1">⚠️ ALERTA DE LIMITE</span>
          <span class="text-[10px] bg-amber-200 text-amber-900 px-1.5 py-0.5 rounded font-bold">${item.perc.toFixed(1)}% do Custo</span>
        </div>
        <p class="mt-1 text-amber-950 font-medium">
          O setor <strong>${item.depto}</strong> aproximou-se do teto de 29%. Restam apenas R$ ${formatarMoedaBR(item.teto - item.gasto)} para novos pedidos.
        </p>
      `;
      container.appendChild(card);
    });
  }

  if (estourados.length === 0 && atencao.length === 0) {
    const card = document.createElement('div');
    card.className = "p-3 rounded-lg bg-emerald-50 border border-emerald-200 text-xs text-emerald-800 shadow-sm";
    card.innerHTML = `
      <div class="flex items-center gap-1.5 font-bold text-emerald-700">
        <span>✅</span> OPERAÇÃO ORÇAMENTÁRIA SAUDÁVEL
      </div>
      <p class="mt-1 text-emerald-900">
        Todos os setores estão operando dentro do teto previsto. Nenhuma ação de bloqueio ou remanejamento necessária no momento.
      </p>
    `;
    container.appendChild(card);
  }
}
